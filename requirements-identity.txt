# Stage 2 (identity) dependencies -- kept separate from requirements.txt and
# requirements-ml.txt because they're only needed once identity/registration
# work begins, and (for the face model, added later) are a heavy download.
#
# Install with:
#   .venv\Scripts\python.exe -m pip install -r requirements-identity.txt
#
# AES-256-GCM authenticated encryption for embeddings + profile fields.
cryptography==49.0.0

# Argon2id password-based key derivation (operator passphrase -> key).
argon2-cffi==25.1.0

# Face model (Stage 2.2+): SCRFD detection + ArcFace embeddings (buffalo_l pack,
# auto-downloaded to ~/.insightface on first use). InsightFace CODE is MIT but
# the pretrained buffalo_l MODELS are non-commercial/research use only -- see
# docs/model_licenses.md.
insightface==1.0.1

# ONNX runtime for the face model. This is the CPU build -- fine for the
# one-time registration flow. For GPU inference in the live loop, swap for
# `onnxruntime-gpu` (must match the installed CUDA/cuDNN, same care as the
# torch CUDA wheels in requirements-ml.txt) and set face.device: "cuda".
onnxruntime==1.27.0
